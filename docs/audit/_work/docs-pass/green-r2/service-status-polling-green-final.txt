**Polling belongs to your app.** Here's the consumer-side snippet the component was designed for:

```tsx
"use client";
import { useEffect, useState } from "react";
import {
  ServiceStatusBadge,
  ServiceStatusBadgeStatus,
} from "@dooph-software/design-system";

export function LiveServiceStatus({ url }: { url: string }) {
  const [status, setStatus] = useState<ServiceStatusBadgeStatus>(
    ServiceStatusBadgeStatus.operational,
  );

  useEffect(() => {
    const poll = async () => {
      try {
        const res = await fetch(url, { cache: "no-store" });
        if (res.ok) {
          const data: { status: ServiceStatusBadgeStatus } = await res.json();
          setStatus(data.status);
        }
      } catch {
        // keep the last status; how an outage looks is the app's call
      }
    };
    poll();
    const id = setInterval(poll, 30_000);
    return () => clearInterval(id);
  }, [url]);

  return <ServiceStatusBadge status={status} />;
}
```
